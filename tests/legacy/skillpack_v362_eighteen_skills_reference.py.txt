class Providers:
    def __init__(self):self.calls=[]
    async def request(self,name,**kwargs):
        self.calls.append((name,kwargs))
        if name=='geocode':
            place=kwargs['place']
            if place=='Springfield':
                return ProviderResult({'results':[
                    {'name':'Springfield','country':'United States','latitude':39.7,'longitude':-89.6},
                    {'name':'Springfield','country':'Australia','latitude':-27.6,'longitude':152.9}]},
                    [Evidence(provider='mock-geocoder')])
            return ProviderResult({'results':[{'name':place,'country':'Japan','latitude':34.7,'longitude':135.5}]},
                                  [Evidence(provider='mock-geocoder')])
        if name=='places':
            kind=kwargs['kind']
            venue={'restaurant':{'name':'Mapped Dining Venue','cuisine':'Japanese'},
                   'hotel':{'name':'Mapped Hotel'},'tourism':{'name':'Mapped Museum'}}.get(kind,{'name':'Mapped Viewpoint'})
            return ProviderResult({'results':[{**venue,'latitude':34.7,'longitude':135.5,'osm_id':123}]},
                                  [Evidence(provider='mock-map')])
        if name=='walking_route':
            return ProviderResult({'distance_m':5400,'duration_s':2400,'geometry':{'type':'LineString','coordinates':[]}},
                                  [Evidence(provider='mock-router')])
        if name=='weather':
            return ProviderResult({'timezone':'Asia/Tokyo','hourly':{'time':['2026-10-20T08:00'],
                   'temperature_2m':[16], 'precipitation_probability':[30]}},
                   [Evidence(provider='mock-weather')])
        raise ProviderUnavailable('Offline fixture has no other capabilities.')

class Store:
    def __init__(self):self.entries=[]
    def add_expense(self,user,session,amount,currency,category):
        self.entries.append({'user':user,'session':session,'amount':amount,
                             'currency':currency,'category':category})
    def expenses(self,user):return [x for x in self.entries if x['user']==user]

@pytest.mark.parametrize('name',sorted(ACTIVE))
@pytest.mark.asyncio
async def test_each_skill_executes_its_documented_example(name,ctx):
    folder=ROOT/'plugins/skills'/name
    spec=SkillRegistry(ROOT/'plugins/skills').get(name)
    assert spec.entrypoint=='handler.py:run'
    assert len((folder/'SKILL.md').read_text().splitlines())>=40
    assert '## Execution protocol' in (folder/'SKILL.md').read_text()
    args=json.loads((folder/'references/input_example.json').read_text())
    result=await SkillRegistry(ROOT/'plugins/skills').run(name,ctx,args)
    assert result.status in {'ok','partial'}, (name,result)
    assert isinstance(result.data,dict) and result.data, (name,result)
    assert result.text and isinstance(result.limitations,list)

def test_installer_is_idempotent_and_archives_only_legacy(tmp_path):
    root=tmp_path
    for name in ACTIVE:
        folder=root/'plugins'/'skills'/name
        folder.mkdir(parents=True)
        (folder/'handler.py').write_text('async def run(ctx,args): pass')
        (folder/'SKILL.md').write_text('Specification')
        (folder/'manifest.json').write_text(json.dumps({'id':name,'entrypoint':'handler.py:run','release_state':'implemented'}))
    retired=root/'plugins'/'skills'/'obsolete'
    retired.mkdir()
    (retired/'manifest.json').write_text('{}')
    first=migrate(root=root)
    assert first['retired_now']==1
    assert (root/'plugins/retired-skills-v361/obsolete/manifest.json').is_file()
    assert migrate(root=root)['retired_now']==0
    assert set(p.parent.name for p in (root/'plugins/skills').glob('*/manifest.json'))==ACTIVE

@pytest.mark.asyncio
async def test_optional_provider_transit_and_place_details(ctx):
    class Extended(Providers):
        async def request(self, capability, **kwargs):
            if capability=='transit_route':
                return ProviderResult({'duration_s':1800,'distance_m':15000},[Evidence(provider='fixture-transit')])
            if capability=='place_details':
                return ProviderResult({'results':[{'place_id':kwargs['place_id'],
                   'name':'Example Museum','opening_hours':'Mo-Fr 10:00-17:00'}]},
                   [Evidence(provider='fixture-place-details')])
            return await super().request(capability, **kwargs)
    ctx.providers=Extended()
    registry=SkillRegistry(ROOT/'plugins/skills')
    points=[{'latitude':35.0,'longitude':135.0},{'latitude':35.1,'longitude':135.1}]
    transit=await registry.run('public-transit-router',ctx,{'waypoints':points})
    assert transit.data['provider_route']['duration_s']==1800
    assert transit.sources[0].provider=='fixture-transit'
    airport=await registry.run('airport-transfer',ctx,{'origin':'Airport','destination':'Hotel','waypoints':points})
    assert airport.data['provider_route']['distance_m']==15000
    details=await registry.run('place-detail-verification',ctx,{'place':'Example Museum',
                'place_id':'abcd1234567890','candidates':[]})
    assert details.data['provider_details'][0]['name']=='Example Museum'
    assert details.data['verification_level']=='provider_details_candidate'
